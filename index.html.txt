<script>
    // Массив слов с их сдвигом относительно центрального вертикального слова (ФАЛЬШИВОМОНЕТНИЧЕСТВО)
    // Строка состоит из: [номер, слово, индекс буквы, которая пересекается с главной вертикалью]
    const words = [
        [1, "ФОЛИАНТ", 0],
        [2, "УРОЖАЙ", 3],
        [3, "КОЗЛЫ", 3],
        [4, "ЦЕРКОВЬ", 4],
        [5, "ПАСТУШЕСТВО", 5],
        [6, "ПШЕНИЦА", 4],
        [7, "ЧРЕВОУГОДИЕ", 5],
        [8, "РОМ", 1],
        [9, "РУМИОН", 3],
        [10, "ЯБЛОКИ", 3],
        [11, "ПОШЛИНА", 4],
        [12, "ПАРТНЕР", 3],
        [13, "ПРЯНОСТИ", 5],
        [14, "ПАЛОМНИЧЕСТВО", 8],
        [15, "ТРАКТИР", 5],
        [16, "ЕПИСКОП", 4]
    ];

    const maxLeft = Math.max(...words.map(w => w[2]));
    const maxRight = Math.max(...words.map(w => w[1].length - 1 - w[2]));
    const totalCols = maxLeft + 1 + maxRight;

    const grid = document.getElementById('grid');

    words.forEach((wordData) => {
        const [num, word, mainIdx] = wordData;
        const row = document.createElement('div');
        row.className = 'row';

        const startSpaces = maxLeft - mainIdx;

        for (let i = 0; i < totalCols; i++) {
            if (i < startSpaces || i >= startSpaces + word.length) {
                const space = document.createElement('div');
                space.className = 'cell-space';
                row.appendChild(space);
            } else {
                const charIdx = i - startSpaces;
                const cellContainer = document.createElement('div');
                cellContainer.className = 'cell-container';

                if (charIdx === 0) {
                    const label = document.createElement('span');
                    label.className = 'num-label';
                    label.innerText = num;
                    cellContainer.appendChild(label);
                }

                const input = document.createElement('input');
                input.className = 'cell';
                input.maxLength = 1;
                input.dataset.answer = word[charIdx];
                
                if (charIdx === mainIdx) {
                    input.classList.add('main-col');
                }

                // Автопереход на следующую ячейку при вводе
                input.addEventListener('input', function() {
                    if (this.value.length === 1) {
                        const next = this.parentElement.nextElementSibling?.querySelector('input');
                        if (next) next.focus();
                    }
                });

                cellContainer.appendChild(input);
                row.appendChild(cellContainer);
            }
        }
        grid.appendChild(row);
    });

    function checkCrossword() {
        const inputs = document.querySelectorAll('.cell');
        let allCorrect = true;

        inputs.forEach(input => {
            if (input.value.toUpperCase() === input.dataset.answer) {
                input.classList.add('correct');
            } else {
                input.classList.remove('correct');
                if (input.value !== '') allCorrect = false;
            }
        });

        if (allCorrect && Array.from(inputs).every(i => i.value !== '')) {
            alert('Поздравляем! Кроссворд решен верно! 🎉');
        }
    }
</script>
